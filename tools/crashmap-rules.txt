# crashmap-rules.txt -- what the crash guard's step 1b (common/CrashStep.h) turns off when a SpringBoard crash points at one of our parts.
# Read by tools/make-crashmap.py at package time; the result (CrashMap.txt) ships next to the parts.
#
# target <name> pref <domain> <key> <off> <default> <row title>|<where>   a switch in Settings: set to <off> (0/1); <default> = its value when
#                                                     never set; the row's title and where the crash was, for the guard's explanation
# target <name> part                                  the part that crashed is not loaded in SpringBoard any more (until the guard's note is cleared)
# image <Image> <target> [map]                        what a crash in that part turns off; "map": a function map is built (the target is then the
#                                                     fallback for code no rule below names)
# sym <Image> <regex> <target>                        a function whose symbol name matches (Logos hooks carry their %group: $MacBanners$...)
# sec <Image> <regex> <target>                        a function whose first line sits under a matching section header (// ==== ... / // ---- ... /
#                                                     %group) in its source file; an "// ===== end ..." header ends a section
# file <Image> <regex> <target>                       a function from a matching source file
# Order: sym rules, then sec, then file, then the image's target; the first match wins. A switch that is already off falls back to "part".
# Every rule must match at least one function (make-crashmap.py stops the build otherwise): keep the section headers the sec rules name when
# comments are edited, or change the rule with them.
#
# Whenever a crash cannot be pinned on one of these (only Apple code on the stack, an unreadable report, one of the two loaders), the guard goes on
# to its step 2 (the whole tweak off) as before.

target part part
# transparent: a pass-through hook (-[UIApplication sendEvent:], -[UIResponder pressesBegan:withEvent:]): on nearly every touch and key stack,
# so it never decides a crash by itself (common/CrashBlame.h): any other non-Apple code below it is blamed instead, and step 1b passes over it

# ---- the status bar (MacStatusBarCore): by feature; anything else = Stock status bar mode ----
image MacStatusBarCore stock map
# Finder (Finder.h, NativeWindow.h, and its glue in StatusBar.x): Finder's own switch -- not the whole status bar. Its SpringBoard pass-through hooks
# (sendEvent: / pressesBegan:) are transparent like the others'; its icon-tap hook is Finder's.
sym MacStatusBarCore \$DMNativeHooks\$SpringBoard\$(sendEvent|pressesBegan) transparent
sym MacStatusBarCore \$DMNativeHooks\$ finder
sym MacStatusBarCore \$DMNativeSysGestures\$ finder
sym MacStatusBarCore ^_?DM(Finder|Native|OpenFinder|OpenNativeWindow)[A-Z] finder
file MacStatusBarCore ^(Finder|NativeWindow)\.h$ finder
# the text window (TextWindow.h) is Finder's; the desktop (Desktop.h, DesktopPlace.h) and the Wi-Fi menu (WiFiMenu.h) have their own switches
file MacStatusBarCore ^TextWindow\.h$ finder
sym MacStatusBarCore ^_?DMDesktop[A-Z] desktopIcons
file MacStatusBarCore ^Desktop\.h$ desktopIcons
file MacStatusBarCore ^DesktopPlace\.h$ desktopIcons
sym MacStatusBarCore ^_?DMWiFi[A-Z] wifiMenu
file MacStatusBarCore ^WiFiMenu\.h$ wifiMenu
# Spotlight (statusbar/SpotlightFiles.x): the tweak settings part (its own functions, its PreferenceLoader watcher, the shared index / donation headers), then the files part

file MacStatusBarCore ^Tweak(Index|Spotlight)\.h$ spotlightTweakSettings
file MacStatusBarCore ^SpotlightFiles\.x$ spotlightFiles
# our sections in Spotlight (SpotlightBridge.h, 1.4.3): the window list has its own switch, everything else is the actions'
sym MacStatusBarCore ^_*DMSpot(WindowRows|Where|OnDesktopThen) spotlightWindows
file MacStatusBarCore ^SpotlightBridge\.h$ spotlightActions
# the Mac Switcher (MacSwitcher.h, 1.4; its Stage Manager desktops MacSwitcherSM.h and Home button MacSwitcherButton.h): its own switch -- the
# functions and their blocks (a block's symbol is the function's name with "___" before it, as in 1.3.6 R3-L3: in the 1.4~int6 release map every
# block of MacSwitcherSM.h and MacSwitcherButton.h -- the desktop switch's, the notice's, the drag's move, the Home double press -- fell to Stock
# status bar mode, the whole Mac status bar; MacSwitcher.h's were caught by its file rule)
sym MacStatusBarCore ^_?DMMSWPressesBegan$ transparent   # (its key pass-through is on every key press stack)
sym MacStatusBarCore ^_*DMMSW[A-Z] macSwitcher
file MacStatusBarCore ^MacSwitcher(SM|Button)?\.h$ macSwitcher

# the Stage Manager engine (its functions, its hook groups, its source sections): windowing off -- the engine stops, the Mac status bar stays
# (unnamed, it fell to the image's target: Stock status bar mode, which stopped the engine too but also took the menu bar; iPadOS 17 runs the
# engine's 17 layout code untested, so a crash there must cost the engine, not the bar)
# (also the blocks inside those functions: a block's symbol is the function's name with "___" before it and "_block_invoke[_N]" after it)
sym MacStatusBarCore ^_*DMSM[A-Z0-9] windowingEnabled
# Guided Access (1.3.7): the engines' windows put away for a session are window code; the rest (the session watcher, its state, the lights' and
# the gates' part) has no switch of its own and runs whatever the status bar style ("stock" would not stop it): the part goes
sym MacStatusBarCore ^_*DMGAWindowsApply windowingEnabled
sym MacStatusBarCore ^_*(DMGA[A-Z]|DMGuidedAccessChanged|DMWatchGuidedAccess|MSBDGuidedAccess) part
# (force quit: the window engines close the app's window first -- the Apple menu's Force Quit and, since 1.3.5, the app icons' menus through
#  MSBDForceQuitApp; and the windowed apps the Dock asks about. Their sections fell to "part" / "traffic lights" before, 1.3.5 logic test R2-L1;
#  the force quit's own blocks -- the kill and the switcher clean-up, run after the window closed -- too, 1.3.5 round 2 R3-L3)
sym MacStatusBarCore ^_*(DMForceQuitBundle(_block_invoke(_[0-9]+)?)?|MSBDForceQuitApp|MSBDAppsWithWindows)$ windowingEnabled
# (the wait of window changes for a menu to close -- also for SpringBoard's own icon menus since 1.3.6 (R3-L1), whose 2 s cap asks SpringBoard to
#  close them: its functions and blocks fell to "part" (the whole status bar off), 1.3.6 logic test L-3)
